# server crash). develop keeps the relay as the activator and checks for NULL: nobody gets the text, the map goes on.

# covers: game_text:1 trigger_relay.delay

frames 3

frames 15
expectn 1 MSTR 23 ->#1: text_all

frames 15
expectnot MSTR 23 ->#1: text_one

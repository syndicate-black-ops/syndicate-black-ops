# env_explosion (FGD): Magnitude sets the fireball (TE_EXPLOSION) and smoke (TE_SMOKE) scale ((magnitude - 50) * 0.6:
# 100 -> 30, 200 -> 90) and hurts what is near (a breakable 40 units away breaks); flags 1 No Damage (the breakable
# stays), 2 Repeatable (a second trigger explodes again; without it the entity is gone after one), 4 No Fireball
# (the TE_EXPLOSION carries scale 0: the sound only), 8 No Smoke, 32 No Sparks (no TE_SPARKS)
# covers: env_explosion env_explosion.imagnitude env_explosion:1 env_explosion:2 env_explosion:4 env_explosion:8 env_explosion:32
map co_fuzz
brushsize 32 32 32
spawn env_explosion targetname e100 iMagnitude 100 origin "1000 0 0"
spawn env_explosion targetname e200 iMagnitude 200 origin "2000 0 0" spawnflags 32
spawn func_breakable model *1 health 10 targetname b200 origin "2040 0 0"
spawn env_explosion targetname enodmg iMagnitude 100 origin "3000 0 0" spawnflags 33
spawn func_breakable model *2 health 10 targetname b3000 origin "3040 0 0"
spawn env_explosion targetname erep iMagnitude 100 origin "4000 0 0" spawnflags 34
spawn env_explosion targetname enofire iMagnitude 100 origin "5000 0 0" spawnflags 36
spawn env_explosion targetname enosmoke iMagnitude 100 origin "6000 0 0" spawnflags 40
activate
connect 1 P1
frames 3
mark explode
fire e100
fire e200
fire enodmg
fire erep
fire enofire
fire enosmoke
frames 5
expectre MSG svc23 ->area: 3 1000 0 0 \d+ 30 15 0$
expectre MSG svc23 ->area: 5 1000 0 0 \d+ 30 12$
expectre MSG svc23 ->area: 3 2000 0 0 \d+ 90 15 0$
expectre MSG svc23 ->area: 5 2000 0 0 \d+ 90 12$
expectdump b3000 health=10.0
expectcount func_breakable 1
expectre MSG svc23 ->area: 3 5000 0 0 \d+ 0 15 0$
expectre MSG svc23 ->area: 5 5000 0 0
expectre MSG svc23 ->area: 3 6000 0 0
expectnotre MSG svc23 ->area: 5 6000 0 0
expectre MSG svc23 ->area: 3 1000 0 0 \d+ 30 15 0$
expectnotre MSG svc23 ->area: 9 2000 0 0
frames 10
mark again
fire e100
fire erep
frames 5
expectn 2 MSG svc23 ->area: 3 4000 0 0
expectn 1 MSG svc23 ->area: 3 1000 0 0

# co_spells: garg_relay (trigger_relay, delay 4, triggerstate 1, fire once) -> multisource teleporter_lock = master of
# the wizard-tower trigger_teleport (here a trigger_once with the same master, so the touch shows in the console).
# The delayed Use comes from a temporary DelayedUse entity, not a member: it toggles the relay's slot, the
# multisource fires and the master opens.
map co_fuzz
connect 1
spawn multisource targetname teleporter_lock target ms_fired
spawn trigger_relay targetname garg_relay target teleporter_lock delay 4 triggerstate 1 spawnflags 1
spawn trigger_relay targetname ms_fired target ms_fired_sink
spawn trigger_once model *112 master teleporter_lock target teleport_used delay 0
spawn trigger_relay targetname teleport_used target sink
activate
frames 3
use 1 world
frames 60
touch 3 1
frames 3
expect Firing: (ms_fired)
expect Firing: (teleport_used)
expectn 0 MultiSrc:Used by non member
